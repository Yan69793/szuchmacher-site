const prices    = { gold: null, silver: null, platinum: null, btc: null, usd: null };
const prevPrices= { gold: null, silver: null, platinum: null, btc: null, usd: null };
// Track last successful fetch time per metal to detect stale data
const lastFetch = { gold: 0, silver: 0, platinum: 0 };

function fmt(n, decimals = 2) {
    return n.toLocaleString('pt-BR', { minimumFractionDigits: decimals, maximumFractionDigits: decimals });
}

function setPrice(id, value) {
    const el = document.getElementById(id);
    if (!el) return;
    el.textContent = value;
    el.classList.remove('flash'); void el.offsetWidth; el.classList.add('flash');
}

function setChange(id, pct) {
    const el = document.getElementById(id);
    if (!el) return;
    el.textContent = (pct >= 0 ? '+' : '') + fmt(pct) + '%';
    el.className = 'ticker-change ' + (pct > 0 ? 'change-up' : pct < 0 ? 'change-down' : 'change-neutral');
}

function applyMetal(key, value, pct) {
    if (!value || isNaN(value) || value <= 0) return;
    const map = {
        gold:     { hero: 'h-gold-price',   heroChg: 'h-gold-change',   sim: 'sp-gold-price',   simChg: 'sp-gold-change' },
        silver:   { hero: 'h-silver-price',  heroChg: 'h-silver-change',  sim: 'sp-silver-price',  simChg: 'sp-silver-change' },
        platinum: { hero: 'h-plat-price',    heroChg: 'h-plat-change',    sim: 'sp-plat-price',    simChg: 'sp-plat-change' },
    };
    const m = map[key];
    if (!m) return;
    prevPrices[key] = prices[key];
    prices[key] = value;
    lastFetch[key] = Date.now();
    const displayPct = (pct !== null && pct !== undefined) ? pct
        : (prevPrices[key] ? ((value - prevPrices[key]) / prevPrices[key]) * 100 : 0);
    setPrice(m.hero, '$ ' + fmt(value));
    setPrice(m.sim,  '$ ' + fmt(value));
    setChange(m.heroChg, displayPct);
    const sc = document.getElementById(m.simChg);
    if (sc) {
        sc.textContent = (displayPct >= 0 ? '+' : '') + fmt(displayPct) + '%';
        sc.className = 'sim-preco-change ' + (displayPct > 0 ? 'change-up' : displayPct < 0 ? 'change-down' : 'change-neutral');
    }
}

// ─────────────────────────────────────────────────────────────
// METALS — estratégia em 5 camadas com retry agressivo
// Cada fonte tenta retornar os 3 metais; se uma falhar parcialmente,
// a próxima camada cobre apenas os metais ainda ausentes.
// ─────────────────────────────────────────────────────────────
async function fetchMetals() {
    const needed = () => ({
        gold:     !prices.gold     || (Date.now() - lastFetch.gold)     > 180000,
        silver:   !prices.silver   || (Date.now() - lastFetch.silver)   > 180000,
        platinum: !prices.platinum || (Date.now() - lastFetch.platinum) > 180000,
    });

    // ── FONTE 1: metals.live (especializada em metais spot) ──
    try {
        const res = await fetch('https://api.metals.live/v1/spot/gold,silver,platinum', { cache: 'no-store' });
        if (res.ok) {
            const data = await res.json();
            // API retorna array de objetos: [{gold: 2800}, {silver: 31.5}, {platinum: 960}]
            data.forEach(item => {
                if (item.gold     && needed().gold)     applyMetal('gold',     parseFloat(item.gold),     null);
                if (item.silver   && needed().silver)   applyMetal('silver',   parseFloat(item.silver),   null);
                if (item.platinum && needed().platinum) applyMetal('platinum', parseFloat(item.platinum), null);
            });
            const n = needed();
            if (!n.gold && !n.silver && !n.platinum) { console.log('[metals.live] OK'); return; }
        }
    } catch(e) { console.warn('[metals.live]', e.message); }

    // ── FONTE 2: Coinbase (suporta XAU, XAG, XPT) ──
    try {
        const n = needed();
        const pairs = [];
        if (n.gold)     pairs.push(['gold',     'XAU-USD']);
        if (n.silver)   pairs.push(['silver',   'XAG-USD']);
        if (n.platinum) pairs.push(['platinum', 'XPT-USD']);

        const results = await Promise.allSettled(
            pairs.map(([, sym]) => fetch(`https://api.coinbase.com/v2/prices/${sym}/spot`, { cache: 'no-store' }))
        );

        for (let i = 0; i < pairs.length; i++) {
            const r = results[i];
            if (r.status === 'fulfilled' && r.value.ok) {
                const d = await r.value.json();
                const val = parseFloat(d.data?.amount);
                if (val > 0) applyMetal(pairs[i][0], val, null);
            }
        }
        const n2 = needed();
        if (!n2.gold && !n2.silver && !n2.platinum) { console.log('[coinbase] OK'); return; }
    } catch(e) { console.warn('[coinbase metals]', e.message); }

    // ── FONTE 3: goldprice.org (spot em USD) ──
    try {
        const res = await fetch('https://data-asg.goldprice.org/dbXRates/USD', { cache: 'no-store' });
        if (res.ok) {
            const data = await res.json();
            const item = data.items?.[0];
            if (item) {
                const n = needed();
                if (n.gold     && item.xauPrice) applyMetal('gold',     parseFloat(item.xauPrice), null);
                if (n.silver   && item.xagPrice) applyMetal('silver',   parseFloat(item.xagPrice), null);
                if (n.platinum && item.xptPrice) applyMetal('platinum', parseFloat(item.xptPrice), null);
                const n2 = needed();
                if (!n2.gold && !n2.silver && !n2.platinum) { console.log('[goldprice.org] OK'); return; }
            }
        }
    } catch(e) { console.warn('[goldprice.org]', e.message); }

    // ── FONTE 4: CoinGecko (PAXG para ouro spot; silver token) ──
    // Prata e platina: usa tokens de referência — menos preciso mas melhor que nada
    try {
        const n = needed();
        const ids = [];
        if (n.gold)   ids.push('pax-gold');
        if (n.silver) ids.push('silver');
        // platina não tem token confiável no CoinGecko; fonte 5 cobre
        if (ids.length > 0) {
            const res = await fetch(
                `https://api.coingecko.com/api/v3/simple/price?ids=${ids.join(',')}&vs_currencies=usd&include_24hr_change=true`,
                { cache: 'no-store' }
            );
            if (res.ok) {
                const data = await res.json();
                if (n.gold && data['pax-gold']?.usd)
                    applyMetal('gold', data['pax-gold'].usd, data['pax-gold'].usd_24h_change || 0);
                if (n.silver && data['silver']?.usd)
                    applyMetal('silver', data['silver'].usd, data['silver'].usd_24h_change || 0);
                console.log('[coingecko] partial OK');
            }
        }
    } catch(e) { console.warn('[coingecko metals]', e.message); }

    // ── FONTE 5: SwissQuote XPT/USD ──
    try {
        const n = needed();
        if (n.platinum) {
            const res = await fetch('https://forex-data-feed.swissquote.com/public-quotes/bboquotes/instrument/XPT/USD', { cache: 'no-store' });
            if (res.ok) {
                const data = await res.json();
                const entry = Array.isArray(data) ? data[0] : data;
                const price = entry?.spreadProfilePrices?.[0]?.ask
                           || entry?.spreadProfilePrices?.[0]?.bid
                           || entry?.ask || entry?.bid;
                if (price && parseFloat(price) > 200) {
                    applyMetal('platinum', parseFloat(price), null);
                    console.log('[swissquote] platinum OK:', parseFloat(price));
                }
            }
        }
    } catch(e) { console.warn('[swissquote]', e.message); }

    // ── FONTE 6: Yahoo Finance futures PL=F — endpoint público para platina ──
    try {
        const n = needed();
        if (n.platinum) {
            const res = await fetch(
                'https://query1.finance.yahoo.com/v8/finance/chart/PL=F?interval=1d&range=1d',
                { cache: 'no-store', headers: { 'Accept': 'application/json' } }
            );
            if (res.ok) {
                const data = await res.json();
                const price = data?.chart?.result?.[0]?.meta?.regularMarketPrice;
                if (price && parseFloat(price) > 200) {
                    applyMetal('platinum', parseFloat(price), null);
                    console.log('[yahoo] platinum OK:', parseFloat(price));
                }
            }
        }
    } catch(e) { console.warn('[yahoo platinum]', e.message); }

    // Log quais metais ainda ausentes após todas as fontes
    const fin = needed();
    if (fin.platinum) {
        console.warn('[metals] platina indisponível — todas as fontes falharam ou bloquearam por CORS');
    }
    if (fin.gold || fin.silver) {
        console.warn('[metals] ainda ausentes:', Object.keys(fin).filter(k => fin[k] && k !== 'platinum'));
    }
}

                console.log('[binance] BTC=' + val);
                return;
            }
        }
    } catch(e) { console.warn('[binance btc]', e.message); }

    // Fonte 3: Coinbase BTC
    try {
        const res = await fetch('https://api.coinbase.com/v2/prices/BTC-USD/spot', { cache: 'no-store' });

                setPrice('sp-btc-price', '$ ' + fmt(val, 0));
                console.log('[coinbase] BTC=' + val);
            }
        }
    } catch(e) { console.warn('[coinbase btc]', e.message); }
}

// ── CALCULATION ENGINE — fórmula correta de juros compostos ──
// Aportes mensais convertidos para taxa mensal equivalente
// FV = PV*(1+r)^n + PMT * [((1+r)^n - 1) / r]
// onde r = taxa mensal = (1 + taxa_anual)^(1/12) - 1
function calcSerie(inicial, aporte, taxaAnual, prazo) {
    const anos = parseInt(prazo) || 10;
    const r    = Math.pow(1 + taxaAnual, 1 / 12) - 1; // taxa mensal equivalente
    const serie = [];
    for (let y = 0; y <= anos; y++) {
        const n   = y * 12; // número de meses
        const fv  = inicial * Math.pow(1 + r, n)
                  + (r > 0
                      ? aporte * (Math.pow(1 + r, n) - 1) / r
                      : aporte * n);
        serie.push(Math.max(fv, 0));
    }
    return serie;
}

function calcFinal(inicial, aporte, taxa, prazo) {
    const s = calcSerie(inicial, aporte, taxa, prazo);
    return s[s.length - 1];
}

function fmtUSD(v) {
    if (v >= 1e6) return '$ ' + (v/1e6).toFixed(2).replace('.', ',') + ' M';
    return '$ ' + fmt(v, 0);
}

function fmtRet(final, inicial, aporte, prazo) {
    const total = inicial + aporte * 12 * prazo;
    const ret = ((final - total) / total) * 100;
    return (ret >= 0 ? '+' : '') + fmt(ret, 1) + '%';
}

function recalcSim(asset) {
    const c = simConfigs[asset];
    if (!c) return;
    const ini  = parseFloat(document.getElementById(c.ini)?.value)   || 10000;
    const ap   = parseFloat(document.getElementById(c.ap)?.value)    || 500;
    const prz  = parseFloat(document.getElementById(c.prazo)?.value) || (asset === 'btc' ? 4 : 10);
    const totalInv = ini + ap * 12 * prz;

    const base = calcFinal(ini, ap, c.base, prz);
    const otim = calcFinal(ini, ap, c.otim, prz);
    const pess = asset === 'btc' ? totalInv * 0.40 : calcFinal(ini, ap, c.pess, prz);

    const p = c.prefix;

    // Valor final da carteira
    document.getElementById(p + '-pess').textContent   = fmtUSD(pess);
    document.getElementById(p + '-base').textContent   = fmtUSD(base);
    document.getElementById(p + '-otim').textContent   = fmtUSD(otim);

    // Retorno sobre total investido
    document.getElementById(p + '-pess-r').textContent = fmtRet(pess, ini, ap, prz);
    document.getElementById(p + '-base-r').textContent = fmtRet(base, ini, ap, prz);
    document.getElementById(p + '-otim-r').textContent = fmtRet(otim, ini, ap, prz);

    // Breakdown: total investido vs lucro
    const breakdown = (fv) => {
        const lucro = fv - totalInv;
        const sign  = lucro >= 0 ? '+' : '';
        return `Aportado: ${fmtUSD(totalInv)} · Retorno: ${sign}${fmtUSD(lucro)}`;
    };
    const setPessBreak = document.getElementById(p + '-pess-break');
    const setBaseBreak = document.getElementById(p + '-base-break');
    const setOtimBreak = document.getElementById(p + '-otim-break');
    if (setPessBreak) setPessBreak.textContent = breakdown(pess);
    if (setBaseBreak) setBaseBreak.textContent = breakdown(base);
    if (setOtimBreak) setOtimBreak.textContent = breakdown(otim);

    // Preço projetado do ativo (se disponível)
    if (asset !== 'btc') {
        const rates = { pess: c.pess, base: c.base, otim: c.otim };
        const currentPrice = prices[asset === 'gold' ? 'gold' : asset === 'silver' ? 'silver' : 'platinum'];
        if (currentPrice) {
            const priceProjected = (key) => fmtUSD(currentPrice * Math.pow(1 + rates[key], prz));
            const elPP = document.getElementById(p + '-pess-price');
            const elBP = document.getElementById(p + '-base-price');
            const elOP = document.getElementById(p + '-otim-price');
            if (elPP) elPP.textContent = 'Preço projetado: ' + priceProjected('pess') + '/oz';
            if (elBP) elBP.textContent = 'Preço projetado: ' + priceProjected('base') + '/oz';
            if (elOP) elOP.textContent = 'Preço projetado: ' + priceProjected('otim') + '/oz';
        }
    }

    drawSimChart(asset, ini, ap, prz, c, pess, base, otim);
}

function recalcComparador() {
    const prazo = parseInt(document.getElementById('c-prazo')?.value) || 10;
    const ativos = [
        { key: 'gold',     ini: 'c-g-ini', ap: 'c-g-ap', tx: 'c-g-tx', label: 'Ouro',    color: '#c9a84c', dash: [] },
        { key: 'silver',   ini: 'c-s-ini', ap: 'c-s-ap', tx: 'c-s-tx', label: 'Prata',   color: '#d0d0d0', dash: [6,3] },
        { key: 'platinum', ini: 'c-p-ini', ap: 'c-p-ap', tx: 'c-p-tx', label: 'Platina', color: '#8ecfdd', dash: [3,3] },
        { key: 'btc',      ini: 'c-b-ini', ap: 'c-b-ap', tx: 'c-b-tx', label: 'Bitcoin', color: '#f07a40', dash: [10,4] },
    ];
    const labels = [];
    for (let i = 0; i <= prazo; i++) labels.push('Ano ' + i);

    const rawSeries = ativos.map(a => {
        const ini  = parseFloat(document.getElementById(a.ini)?.value) || 10000;
        const ap   = parseFloat(document.getElementById(a.ap)?.value)  || 500;
        const taxa = parseFloat(document.getElementById(a.tx)?.value)  || 8;
        return { ...a, ini, ap, taxa, serie: calcSerie(ini, ap, taxa / 100, prazo) };
    });

    const datasets = rawSeries.map(a => {
        let data;
        if (compModo === 'absoluto') {
            data = a.serie;
        } else if (compModo === 'indice') {
            const base = a.serie[0];
            data = a.serie.map(v => (v / base) * 100);
        } else {
            data = a.serie.map((v, i) => {
                const totalInv = a.ini + a.ap * 12 * i;
                return ((v - totalInv) / totalInv) * 100;
            });
        }
        if (compEscala === 'logarithmic') data = data.map(v => Math.max(v, 0.01));
        return { label: a.label, data, borderColor: a.color, backgroundColor: a.color + '18', borderWidth: 2.5, pointRadius: prazo <= 10 ? 4 : 2, pointHoverRadius: 6, tension: 0.35, fill: false, borderDash: a.dash };
    });

    const canvas = document.getElementById('chart-comparador');
    if (!canvas) return;
    if (compChart) compChart.destroy();

    const yLabel      = compModo === 'absoluto' ? v => fmtUSD(v) : compModo === 'indice' ? v => fmt(v, 1) + ' pts' : v => fmt(v, 1) + '%';
    const tooltipLbl  = compModo === 'absoluto' ? ctx => ` ${ctx.dataset.label}: ${fmtUSD(ctx.raw)}` : compModo === 'indice' ? ctx => ` ${ctx.dataset.label}: ${fmt(ctx.raw, 1)} pts` : ctx => ` ${ctx.dataset.label}: ${fmt(ctx.raw, 1)}%`;

    compChart = new Chart(canvas, {
        type: 'line',
        data: { labels, datasets },
        options: {
            responsive: true, maintainAspectRatio: false,
            interaction: { mode: 'index', intersect: false },
            plugins: {
                legend: { labels: { color: '#aaa', font: { size: 12, family: 'DM Mono' }, boxWidth: 28, padding: 20 }, position: 'top' },
                tooltip: { backgroundColor: '#0d0d0d', borderColor: '#2a2a2a', borderWidth: 1, titleColor: '#777', bodyColor: '#e8e8e8', bodyFont: { family: 'DM Mono', size: 12 }, padding: 12, callbacks: { label: tooltipLbl } }
            },
            scales: {
                x: { ticks: { color: '#555', font: { size: 11, family: 'DM Mono' } }, grid: { color: 'rgba(255,255,255,0.05)' } },
                y: { type: compEscala, ticks: { color: '#555', font: { size: 11, family: 'DM Mono' }, callback: yLabel }, grid: { color: 'rgba(255,255,255,0.05)' }, ...(compModo === 'indice' ? { suggestedMin: 80 } : {}) }
            }
        }
    });

    const tbody = document.getElementById('comp-tbody');
    tbody.innerHTML = rawSeries.map(a => {
        const final = a.serie[a.serie.length - 1];
        const totalInv = a.ini + a.ap * 12 * prazo;
        const ret  = ((final - totalInv) / totalInv) * 100;
        const cagr = (Math.pow(final / Math.max(a.ini, 1), 1 / prazo) - 1) * 100;
        const retColor = ret >= 0 ? '#4caf7d' : '#e05c5c';
        const rank  = rawSeries.slice().sort((x, y) => y.serie[y.serie.length - 1] - x.serie[x.serie.length - 1]).findIndex(x => x.key === a.key) + 1;
        const medal = rank === 1 ? '🥇' : rank === 2 ? '🥈' : rank === 3 ? '🥉' : '';
        return `<tr>
            <td><span style="color:${a.color};font-weight:700;">${a.label}</span> <span style="font-size:14px;">${medal}</span></td>
            <td style="color:var(--muted);font-family:'DM Mono',monospace;">$ ${fmt(totalInv, 0)}</td>
            <td style="color:${a.color};font-family:'DM Mono',monospace;">${fmtUSD(final)}</td>
            <td style="color:${retColor};font-family:'DM Mono',monospace;">${ret >= 0 ? '+' : ''}${fmt(ret,1)}%</td>
            <td style="color:${retColor};font-family:'DM Mono',monospace;">${cagr >= 0 ? '+' : ''}${fmt(cagr,1)}% a.a.</td>
        </tr>`;
    }).join('');
}

    const items = getAllocItems();

    if (donutChart) donutChart.destroy();
    donutChart = new Chart(canvas, {
        type: 'doughnut',
        data: {
            labels: items.map(d => d.label + ' ' + d.pct + '%'),
            datasets: [{
                data: items.map(d => d.pct),
                backgroundColor: items.map(d => d.color),
                borderColor: '#0f0f0f',
                borderWidth: 3,
                hoverBorderWidth: 4,
                hoverBorderColor: '#1a1a1a',
            }]
        },
        options: {
            responsive: true,
            maintainAspectRatio: false,
            cutout: '68%',
            plugins: {
                legend: { display: false },
                tooltip: {
                    backgroundColor: '#111',
                    borderColor: '#2a2a2a',
                    borderWidth: 1,
                    titleColor: '#aaa',
                    bodyColor: '#e8e8e8',
                    bodyFont: { family: 'DM Mono', size: 12 },
                    callbacks: {
                        label: ctx => {
                            const v = (val * ctx.raw) / 100;
                            return ' ' + ctx.raw + '% · R$ ' + fmt(v, 0);
                        }
                    }
                }
            },
            animation: { animateRotate: true, duration: 600 }
        }
    });

    // Atualiza legenda
    const legEl = document.getElementById('donut-legend');
    if (legEl) {
        legEl.innerHTML = items.map(i =>
            `<div class="donut-leg-item"><span class="donut-leg-dot" style="background:${i.color}"></span><span class="donut-leg-label">${i.label} ${i.pct}%</span></div>`
        ).join('');
    }
}

let benchPrazo = 5;
let benchChart = null;

function setBenchPrazo(anos, btn) {
    benchPrazo = anos;
    document.querySelectorAll('.bench-prazo-group .comp-toggle').forEach(b => b.classList.remove('active'));
    btn.classList.add('active');
    desenharBenchmark();

function desenharBenchmark() {
    const val       = parseInt(document.getElementById('portfolio-slider')?.value) || 500000;
    const prazo     = benchPrazo;
    const canvas    = document.getElementById('chart-benchmark');
    const tbody     = document.getElementById('bench-tbody');
    if (!canvas) return;

    // ── PREMISSAS MACROECONÔMICAS ──
    // Selic: parte 15% a.a., converge para 11% em 24 meses (ciclo BCB 2025-2026)
    // IPCA: parte 5,5%, converge para 4% ao longo de 3 anos (meta BCB)
    // NTN-B curta: IPCA + 6,5% (prêmio atual do Tesouro IPCA+ 2027)
    // Taxa CDI ≈ Selic − 0,1%

    const labels = [];
    for (let y = 0; y <= prazo; y++) labels.push(y === 0 ? 'Hoje' : 'Ano ' + y);

    // Selic anual projetada ano a ano
    function selicAno(y) {
        if (y === 0) return 0.150;
        if (y === 1) return 0.132; // queda de ~1.8pp no ano 1
        if (y === 2) return 0.115; // queda de ~1.7pp no ano 2
        return 0.110;              // estabiliza em 11%
    }

    // IPCA anual projetado
    function ipcaAno(y) {
        if (y === 0) return 0.055;
        if (y === 1) return 0.048;
        if (y === 2) return 0.042;
        return 0.040;
    }

    // NTN-B = IPCA + 6,5%
    function ntnbAno(y) { return ipcaAno(y) + 0.065; }

    // Portfólio MultiAsset ponderado (BRL, cenário base)
    // Renda fixa: NTN-B 30% + CDI 25% — já em BRL, não precisa converter
    // Metais em USD: ouro 9,3%, prata 8,5%, btc 40% — ponderados pelos pesos
    // Simplificação: soma ponderada das taxas base convertidas para BRL com câmbio estável
    // (não projetamos variação cambial — investidor considera exposição USD como hedge)
    const pesoRF   = 0.55;  // NTN-B 30% + CDI 25%
    const pesoMet  = 0.20;  // Ouro 15% + Prata 5%
    const pesoBTC  = 0.10;
    const pesoRes  = 0.15;  // Reserva CDI

    function portfolioAno(y) {
        const selic  = selicAno(y);
        const ntnb   = ntnbAno(y);
        // taxa ponderada da renda fixa (NTN-B: 30/55 do bloco RF; CDI: 25/55)
        const rfRate = (ntnb * 30 + selic * 25) / 55;
        const rfTotal= rfRate * pesoRF;
        // metais: média ponderada ouro 9.3%, prata 8.5% — convertendo para BRL assume câmbio flat
        const metRate= (0.093 * 15 + 0.085 * 5) / 20;
        const metTotal = metRate * pesoMet;
        // btc cenário base 40% aa
        const btcTotal = 0.40 * pesoBTC;
        // reserva ≈ CDI
        const resTotal = selic * pesoRes;
        return rfTotal + metTotal + btcTotal + resTotal;
    }

    // Séries acumuladas
    const seriePort  = [val];
    const serieSelic = [val];
    const serieNtnb  = [val];

    for (let y = 1; y <= prazo; y++) {
        seriePort.push( seriePort[y-1]  * (1 + portfolioAno(y)));
        serieSelic.push(serieSelic[y-1] * (1 + selicAno(y)));
        serieNtnb.push( serieNtnb[y-1]  * (1 + ntnbAno(y)));
    }

    if (benchChart) benchChart.destroy();
    benchChart = new Chart(canvas, {
        type: 'line',
        data: {
            labels,
            datasets: [
                {
                    label: 'Portfólio MultiAsset',
                    data: seriePort,
                    borderColor: '#c9a84c',
                    backgroundColor: 'rgba(201,168,76,0.08)',
                    borderWidth: 3,
                    pointRadius: 5,
                    pointHoverRadius: 7,
                    tension: 0.35,
                    fill: true,
                    pointBackgroundColor: '#c9a84c',
                },
                {
                    label: 'NTN-B curta (IPCA+6,5%)',
                    data: serieNtnb,
                    borderColor: '#4caf7d',
                    backgroundColor: 'rgba(76,175,125,0.05)',
                    borderWidth: 2,
                    pointRadius: 4,
                    pointHoverRadius: 6,
                    tension: 0.3,
                    fill: false,
                    borderDash: [6, 3],
                    pointBackgroundColor: '#4caf7d',
                },
                {
                    label: 'Selic projetada',
                    data: serieSelic,
                    borderColor: '#5b9cf6',
                    backgroundColor: 'rgba(91,156,246,0.04)',
                    borderWidth: 2,
                    pointRadius: 4,
                    pointHoverRadius: 6,
                    tension: 0.3,
                    fill: false,
                    borderDash: [3, 3],
                    pointBackgroundColor: '#5b9cf6',
                },
            ]
        },
        options: {
            responsive: true,
            maintainAspectRatio: false,
            interaction: { mode: 'index', intersect: false },
            plugins: {
                legend: {
                    position: 'top',
                    labels: {
                        color: '#aaa',
                        font: { size: 12, family: 'DM Mono' },
                        boxWidth: 28,
                        padding: 16,
                        usePointStyle: false,
                    }
                },
                tooltip: {
                    backgroundColor: '#0d0d0d',
                    borderColor: '#2a2a2a',
                    borderWidth: 1,
                    titleColor: '#777',
                    bodyColor: '#e8e8e8',
                    bodyFont: { family: 'DM Mono', size: 12 },
                    padding: 12,
                    callbacks: {
                        label: ctx => ` ${ctx.dataset.label}: R$ ${fmt(ctx.raw, 0)}`
                    }
                }
            },
            scales: {
                x: {
                    ticks: { color: '#555', font: { size: 11, family: 'DM Mono' } },
                    grid: { color: 'rgba(255,255,255,0.04)' }
                },
                y: {
                    ticks: {
                        color: '#555',
                        font: { size: 11, family: 'DM Mono' },
                        callback: v => fmtBRLShort(v)
                    },
                    grid: { color: 'rgba(255,255,255,0.04)' }
                }
            }
        }
    });

    // Tabela comparativa
    const final = {
        port:  seriePort[prazo],
        ntnb:  serieNtnb[prazo],
        selic: serieSelic[prazo],
    };

    // IPCA acumulado no período para calcular retorno real
    let ipcaAcum = 1;
    for (let y = 1; y <= prazo; y++) ipcaAcum *= (1 + ipcaAno(y));

    function rowData(label, finalVal, color) {
        const ret    = ((finalVal - val) / val) * 100;
        const cagr   = (Math.pow(finalVal / val, 1 / prazo) - 1) * 100;
        const real   = ((finalVal / val) / ipcaAcum - 1) * 100;
        const sign   = ret  >= 0 ? '+' : '';
        const rSign  = real >= 0 ? '+' : '';
        const cSign  = cagr >= 0 ? '+' : '';
        const rColor = ret  >= 0 ? '#4caf7d' : '#e05c5c';
        return `<tr>
            <td><span style="color:${color};font-weight:700;">${label}</span></td>
            <td style="color:${rColor};font-family:'DM Mono',monospace;">${sign}${fmt(ret,1)}%</td>
            <td style="color:${rColor};font-family:'DM Mono',monospace;">${rSign}${fmt(real,1)}%</td>
            <td style="color:${rColor};font-family:'DM Mono',monospace;">${cSign}${fmt(cagr,1)}% a.a.</td>
            <td style="color:${color};font-family:'DM Mono',monospace;">R$ ${fmt(finalVal,0)}</td>
        </tr>`;
    }

    if (tbody) {
        tbody.innerHTML =
            rowData('Portfólio MultiAsset', final.port,  '#c9a84c') +
            rowData('NTN-B curta (IPCA+6,5%)', final.ntnb, '#4caf7d') +
            rowData('Selic projetada',      final.selic, '#5b9cf6');
    }
}
